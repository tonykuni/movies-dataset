  if (-not (Get-Command pwsh -ErrorAction SilentlyContinue)) { winget install --id Microsoft.PowerShell -e --accept-source-agreements --accept-package-agreements --silent | Out-Host }
  if (-not (Get-Command git -ErrorAction SilentlyContinue)) { winget install --id Git.Git -e --accept-source-agreements --accept-package-agreements --silent | Out-Host }
  $env:Path = [Environment]::GetEnvironmentVariable('Path','Machine') + ';' + [Environment]::GetEnvironmentVariable('Path','User')
  if (-not (Test-Path (Join-Path $r '.git'))) { git clone --depth 50 https://github.com/tonykuni/movies-dataset.git $r | Out-Host } else { git -C $r fetch -q origin main; git -C $r merge --no-ff origin/main -m 'boot: merge main' | Out-Host }
  $via = Join-Path $r 'VeritasIntelligenceAnalytics'; $fb = Join-Path $via 'docs\handoff\ai\VIA_DesktopFourBlocks_v0100.py'; $wb = Join-Path $via 'docs\handoff\ai\VIA_WorkbenchLock_v0101.py'
  pwsh -NoProfile -ExecutionPolicy Bypass -Command "& { `$env:VIA_FROM_VCGC='YES'; python '$fb' --root '$r' --blocks 1,2 --install; python '$fb' --root '$r' --blocks 3,4,5 --mode live --jobs 5; python '$wb' all --via '$via' --open }" }
